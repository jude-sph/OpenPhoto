# Travels Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Travels section: trip folders become a spinnable chronological ring of postcards (with a ratchet click), each trip editable (name, folders incl. subfolders, dates, map-recognised places, cover), discovered from a Travels folder or added by hand, stored as sovereign JSON and mirrored to drives.

**Architecture:** Human decisions live in `<libraryRoot>/.openphoto/travels/` (`settings.json` + one `<id>.json` per trip record); a pure `TravelResolver` combines settings + records + the folder tree into trip definitions; the catalog answers derived stats (count, date span, places, default cover) with aggregate queries over `dirPath`; SwiftUI renders the ring, inspector and trip page from `[TravelSummary]` held on `AppState`.

**Tech Stack:** Swift 6 / SwiftUI (macOS 15), GRDB, Swift Testing (`swift test`), `AtomicFile`, `AlbumStore` pattern, GeoNames dataset via `GeoNamesLoader`, `ThumbnailImage` / `MediaTile`, AVFoundation (click sound).

// Sources/OpenPhotoCore/Travels/TravelRecord.swift
import Foundation

/// A place a trip visited. Resolved against the bundled GeoNames city list when the user types it
/// ("map-recognised": coordinates + GeoNames id travel with the record), or free text with every
/// optional nil when nothing matched. See vault-format-v1 §16.
public struct TravelPlace: Codable, Sendable, Equatable, Hashable {
    public var name: String
    public var region: String?
    public var country: String?
    public var countryCode: String?
    public var latitude: Double?
    public var longitude: Double?
    public var geonameID: Int?

    public init(name: String, region: String? = nil, country: String? = nil, countryCode: String? = nil,
                latitude: Double? = nil, longitude: Double? = nil, geonameID: Int? = nil) {
        self.name = name; self.region = region; self.country = country; self.countryCode = countryCode
        self.latitude = latitude; self.longitude = longitude; self.geonameID = geonameID
    }

    /// True when the place carries coordinates (a GeoNames match), false for free text.
    public var isOnMap: Bool { latitude != nil && longitude != nil }
}

/// One trip: the user's decisions only. Everything derivable (count, date span, detected places,
/// default cover) is computed from the catalog at display time and never stored. `nil` on
/// `name`/`startDate`/`endDate`/`places`/`coverHash` means "derive it". `places == []` means
/// "no places" (distinct from nil). Sovereign JSON at `.openphoto/travels/<id>.json`.
public struct TravelRecord: Codable, Sendable, Equatable {
    public var id: String
    public var name: String?
    public var folders: [String]
    public var startDate: String?     // "YYYY-MM-DD"
    public var endDate: String?
    public var places: [TravelPlace]?
    public var coverHash: String?
    public var createdAtMs: Int64
    public var modifiedAtMs: Int64

    public init(id: String, name: String?, folders: [String], startDate: String?, endDate: String?,
                places: [TravelPlace]?, coverHash: String?, createdAtMs: Int64, modifiedAtMs: Int64) {
        self.id = id; self.name = name; self.folders = folders; self.startDate = startDate
        self.endDate = endDate; self.places = places; self.coverHash = coverHash
        self.createdAtMs = createdAtMs; self.modifiedAtMs = modifiedAtMs
    }

    // Encode nil as explicit JSON null so every key is present on disk (readers key on presence).
    private enum K: String, CodingKey {
        case id, name, folders, startDate, endDate, places, coverHash, createdAtMs, modifiedAtMs
    }
    public func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: K.self)
        try c.encode(id, forKey: .id); try c.encode(name, forKey: .name)
        try c.encode(folders, forKey: .folders)
        try c.encode(startDate, forKey: .startDate); try c.encode(endDate, forKey: .endDate)
        try c.encode(places, forKey: .places); try c.encode(coverHash, forKey: .coverHash)
        try c.encode(createdAtMs, forKey: .createdAtMs); try c.encode(modifiedAtMs, forKey: .modifiedAtMs)
    }
    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: K.self)
        id = try c.decode(String.self, forKey: .id)
        name = try c.decodeIfPresent(String.self, forKey: .name)
        folders = try c.decode([String].self, forKey: .folders)
        startDate = try c.decodeIfPresent(String.self, forKey: .startDate)
        endDate = try c.decodeIfPresent(String.self, forKey: .endDate)
        places = try c.decodeIfPresent([TravelPlace].self, forKey: .places)
        coverHash = try c.decodeIfPresent(String.self, forKey: .coverHash)
        createdAtMs = try c.decode(Int64.self, forKey: .createdAtMs)
        modifiedAtMs = try c.decode(Int64.self, forKey: .modifiedAtMs)
    }
}

/// `.openphoto/travels/settings.json`: the Travels folder (direct subfolders become trips when
/// `autoDiscover`), and the discovered subfolders the user removed (`hidden`).
public struct TravelSettings: Codable, Sendable, Equatable {
    public var travelsFolder: String?
    public var autoDiscover: Bool
    public var hidden: [String]
    public init(travelsFolder: String?, autoDiscover: Bool, hidden: [String]) {
        self.travelsFolder = travelsFolder; self.autoDiscover = autoDiscover; self.hidden = hidden
    }
    public static let defaults = TravelSettings(travelsFolder: nil, autoDiscover: true, hidden: [])

    private enum K: String, CodingKey { case travelsFolder, autoDiscover, hidden }
    public func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: K.self)
        try c.encode(travelsFolder, forKey: .travelsFolder)
        try c.encode(autoDiscover, forKey: .autoDiscover)
        try c.encode(hidden, forKey: .hidden)
    }
    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: K.self)
        travelsFolder = try c.decodeIfPresent(String.self, forKey: .travelsFolder)
        autoDiscover = try c.decodeIfPresent(Bool.self, forKey: .autoDiscover) ?? true
        hidden = try c.decodeIfPresent([String].self, forKey: .hidden) ?? []
    }
}
```

Note: `JSONEncoder` drops nil optionals by default; the explicit `encode(_:forKey:)` on an optional writes `null`, which is what the test asserts.

// Sources/OpenPhotoCore/Travels/TravelStore.swift
import Foundation

/// Sovereign trip state in `<libraryRoot>/.openphoto/travels/`: `settings.json` plus one JSON per
/// trip record. Same pattern as `AlbumStore`: atomic + crash-durable writes, a corrupt file loses one
/// record never all, and a one-way mirror to drives. See vault-format-v1 §16.
public enum TravelStore {
    static let settingsFileName = "settings.json"

    public static func directoryURL(libraryRoot: URL) -> URL {
        libraryRoot.appendingPathComponent(Vault.stateDirName).appendingPathComponent("travels")
    }
    private static func fileURL(id: String, libraryRoot: URL) -> URL {
        directoryURL(libraryRoot: libraryRoot).appendingPathComponent(id + ".json")
    }
    private static func settingsURL(libraryRoot: URL) -> URL {
        directoryURL(libraryRoot: libraryRoot).appendingPathComponent(settingsFileName)
    }
    private static var encoder: JSONEncoder {
        let enc = JSONEncoder(); enc.outputFormatting = [.prettyPrinted, .sortedKeys]; return enc
    }

    // MARK: settings

    /// Missing or corrupt settings → `TravelSettings.defaults`.
    public static func loadSettings(libraryRoot: URL) -> TravelSettings {
        guard let data = try? Data(contentsOf: settingsURL(libraryRoot: libraryRoot)),
              let s = try? JSONDecoder().decode(TravelSettings.self, from: data) else { return .defaults }
        return s
    }
    public static func saveSettings(_ s: TravelSettings, libraryRoot: URL) throws {
        try AtomicFile.write(encoder.encode(s), to: settingsURL(libraryRoot: libraryRoot))
    }

    // MARK: records

    /// Every record, skipping `settings.json` and any unreadable/corrupt file. Sorted by id for a
    /// stable order (presentation order is decided by the resolver, by date).
    public static func loadAll(libraryRoot: URL) -> [TravelRecord] {
        let urls = (try? FileManager.default.contentsOfDirectory(
            at: directoryURL(libraryRoot: libraryRoot), includingPropertiesForKeys: nil)) ?? []
        let dec = JSONDecoder()
        var out: [TravelRecord] = []
        for u in urls where u.pathExtension == "json" && u.lastPathComponent != settingsFileName {
            if let data = try? Data(contentsOf: u), let r = try? dec.decode(TravelRecord.self, from: data) {
                out.append(r)
            }
        }
        return out.sorted { $0.id < $1.id }
    }
    public static func load(id: String, libraryRoot: URL) -> TravelRecord? {
        guard let data = try? Data(contentsOf: fileURL(id: id, libraryRoot: libraryRoot)) else { return nil }
        return try? JSONDecoder().decode(TravelRecord.self, from: data)
    }
    public static func save(_ r: TravelRecord, libraryRoot: URL) throws {
        try AtomicFile.write(encoder.encode(r), to: fileURL(id: r.id, libraryRoot: libraryRoot))
    }
    public static func delete(id: String, libraryRoot: URL) {
        try? FileManager.default.removeItem(at: fileURL(id: id, libraryRoot: libraryRoot))
    }

    // MARK: drive mirror

    /// One-way mirror of `travels/` (records AND settings.json) to `<driveStateDir>/travels/`: copy
    /// every source file, delete drive files absent on the source. Paths inside are vault-relative
    /// and drives keep the vault's folder layout, so files copy verbatim.
    public static func syncToDrive(libraryRoot: URL, driveStateDir: URL) throws {
        let fm = FileManager.default
        let srcDir = directoryURL(libraryRoot: libraryRoot)
        let dstDir = driveStateDir.appendingPathComponent("travels")
        let srcFiles = ((try? fm.contentsOfDirectory(at: srcDir, includingPropertiesForKeys: nil)) ?? [])
            .filter { $0.pathExtension == "json" }
        try fm.createDirectory(at: dstDir, withIntermediateDirectories: true)
        let srcNames = Set(srcFiles.map(\.lastPathComponent))
        for f in srcFiles {
            try AtomicFile.write(try Data(contentsOf: f), to: dstDir.appendingPathComponent(f.lastPathComponent))
        }
        let dstFiles = ((try? fm.contentsOfDirectory(at: dstDir, includingPropertiesForKeys: nil)) ?? [])
            .filter { $0.pathExtension == "json" }
        for f in dstFiles where !srcNames.contains(f.lastPathComponent) { try? fm.removeItem(at: f) }
    }
}
```

```markdown
## 16. Travels (`travels/`)

A **trip** is a human-authored record over one or more library folders — a name, the folders it spans (each including its whole subtree), an optional date override, an optional list of places, and an optional cover — that does **not** move or copy files. Trips are **normative, sovereign state**: this directory is the authoritative record; the app derives everything else (photo count, date span, detected places, default cover) from the photos at display time and stores none of it.

- **Location:** `<primary-vault-root>/.openphoto/travels/settings.json` plus one file per trip at `…/travels/<id>.json`, where `<id>` is the trip's UUID and equals the filename stem.
- **Drive propagation:** when OpenPhoto syncs to a canonical/backup drive it mirrors this directory (including `settings.json`) to the drive's `.openphoto/travels/` (copying present files, deleting absent ones). Folder paths are vault-relative and drives keep the vault's folder layout, so the files copy verbatim.

### 16.1 `settings.json`

```json
{
  "travelsFolder": "Travel",
  "autoDiscover": true,
  "hidden": ["Travel/Scans from Mum"]
}
```

- `travelsFolder` — a vault-relative directory path (forward slashes, no leading slash; `""` is the vault root) or `null`.
- `autoDiscover` — when `true` and `travelsFolder` is non-null, every **direct** subfolder of it is a trip unless it is listed in `hidden` or is equal to / nested inside a folder of some trip record (the record supersedes the discovery). A discovered trip has no file until the user edits it.
- `hidden` — direct subfolders of `travelsFolder` the user removed from Travels. Entries that are no longer direct subfolders are ignored.

A missing `settings.json` means `{ "travelsFolder": null, "autoDiscover": true, "hidden": [] }`.

### 16.2 `<id>.json`

```json
{
  "id": "5C2E9F1A-…-UUID",
  "name": "Japan 2026",
  "folders": ["Travel/Japan 2026"],
  "startDate": "2026-04-03",
  "endDate": "2026-04-15",
  "places": [
    { "name": "Kyoto", "region": "Kyoto", "country": "Japan", "countryCode": "JP",
      "latitude": 35.02107, "longitude": 135.75385, "geonameID": 1857910 },
    { "name": "Grandma's village", "region": null, "country": null, "countryCode": null,
      "latitude": null, "longitude": null, "geonameID": null }
  ],
  "coverHash": "sha256:…",
  "createdAtMs": 1789000000000,
  "modifiedAtMs": 1789000000000
}
```

- `id` — UUID string, immutable, equal to the filename stem.
- `name` — display name, or `null` ⇒ derived: the last path component of `folders[0]`.
- `folders` — **non-empty** ordered array of vault-relative directory paths; each includes its entire subtree. No path should equal or lie inside another in the same array (writers normalise; readers tolerate).
- `startDate` / `endDate` — ISO-8601 calendar dates (`YYYY-MM-DD`, local calendar, no time zone) or `null` ⇒ derived from the earliest/latest capture time of the trip's photos. If only one is set, the other is derived. Writers keep `startDate ≤ endDate`.
- `places` — ordered array, or `null` ⇒ derived from the reverse-geocoded cities of the trip's photos (most photos first). Each element has `name` (required) and the GeoNames fields `region`, `country`, `countryCode`, `latitude`, `longitude`, `geonameID` — all `null` together for a free-text place that matched no city. `geonameID` is the integer id in the GeoNames `cities15000` extract. An **empty array** means "no places" and is distinct from `null`.
- `coverHash` — a content hash (§2) or `null` ⇒ derived: the earliest photo of the trip. A cover whose content is absent falls back to the derived cover at display time; the value is kept.
- `createdAtMs` / `modifiedAtMs` — epoch-milliseconds.

Reader obligations: an unreadable/corrupt file MUST be skipped without affecting the others; unknown keys MUST be ignored; a missing `settings.json` MUST be read as the defaults above. Writers MUST write atomically (§10 step 2). Place data is © GeoNames (https://www.geonames.org), CC BY 4.0.
```

// Sources/OpenPhotoCore/Travels/TravelResolver.swift
import Foundation

/// Aggregate facts the catalog derives for a trip's folders (Catalog+Travels.swift).
public struct TravelStats: Sendable, Equatable {
    public var count: Int
    public var minTakenAtMs: Int64?
    public var maxTakenAtMs: Int64?
    public var earliestHash: String?
    public init(count: Int, minTakenAtMs: Int64?, maxTakenAtMs: Int64?, earliestHash: String?) {
        self.count = count; self.minTakenAtMs = minTakenAtMs; self.maxTakenAtMs = maxTakenAtMs
        self.earliestHash = earliestHash
    }
}

/// A trip as the resolver sees it: either a stored record or a folder discovered under the
/// Travels folder (no file yet). `id` of a discovered trip is "folder:<path>" and is never persisted.
public struct TravelDefinition: Sendable, Equatable, Identifiable {
    public enum Source: Sendable, Equatable {
        case record(TravelRecord)
        case discovered(String)
    }
    public var id: String
    public var source: Source
    public var folders: [String]
    public init(id: String, source: Source, folders: [String]) {
        self.id = id; self.source = source; self.folders = folders
    }
    public var record: TravelRecord? { if case .record(let r) = source { return r }; return nil }
    public var isDiscovered: Bool { record == nil }
}

/// What the UI shows for a trip: the record's decisions merged over the derived facts.
public struct TravelSummary: Sendable, Equatable, Identifiable {
    public var id: String
    public var definition: TravelDefinition
    public var name: String
    public var folders: [String]
    public var startDate: String?
    public var endDate: String?
    public var datesOverridden: Bool
    public var places: [TravelPlace]
    public var placesOverridden: Bool
    public var count: Int
    public var coverHash: String?
    public var coverOverridden: Bool

    public var year: Int? { startDate.flatMap { Int($0.prefix(4)) } }
    public var country: String? { places.first?.country }
}

public enum TravelResolver {
    public static let discoveredIDPrefix = "folder:"

    /// Drop duplicates and any path equal to or nested inside another; keep first-seen order.
    public static func normalise(folders: [String]) -> [String] {
        var out: [String] = []
        for f in folders {
            if out.contains(where: { $0 == f || f.hasPrefix($0 + "/") }) { continue }
            out.removeAll { $0.hasPrefix(f + "/") }
            out.append(f)
        }
        return out
    }

    /// Records first (in the given order), then discovered direct subfolders of the Travels folder
    /// (tree order) that are neither hidden nor covered by a record's folder.
    public static func resolve(settings: TravelSettings, records: [TravelRecord],
                               folderTree: [FolderNode]) -> [TravelDefinition] {
        var defs = records.map {
            TravelDefinition(id: $0.id, source: .record($0), folders: normalise(folders: $0.folders))
        }
        guard settings.autoDiscover, let root = settings.travelsFolder,
              let rootNode = find(root, in: folderTree) else { return defs }
        let covered = defs.flatMap(\.folders)
        let hidden = Set(settings.hidden)
        for child in rootNode.children {
            let p = child.path
            if hidden.contains(p) { continue }
            if covered.contains(where: { p == $0 || p.hasPrefix($0 + "/") || $0.hasPrefix(p + "/") }) { continue }
            defs.append(TravelDefinition(id: discoveredIDPrefix + p, source: .discovered(p), folders: [p]))
        }
        return defs
    }

    private static func find(_ path: String, in nodes: [FolderNode]) -> FolderNode? {
        for n in nodes {
            if n.path == path { return n }
            if path.hasPrefix(n.path + "/") || n.path.isEmpty, let hit = find(path, in: n.children) { return hit }
        }
        return nil
    }

    /// Merge the record's decisions over the derived facts (nil ⇒ derived; `places == []` ⇒ none).
    public static func summarise(_ def: TravelDefinition, stats: TravelStats,
                                 detectedPlaces: [TravelPlace]) -> TravelSummary {
        let r = def.record
        let derivedStart = stats.minTakenAtMs.map(dateString(fromMs:))
        let derivedEnd = stats.maxTakenAtMs.map(dateString(fromMs:))
        var start = r?.startDate ?? derivedStart
        var end = r?.endDate ?? derivedEnd
        if let s = start, let e = end, s > e { (start, end) = (e, s) }
        return TravelSummary(
            id: def.id, definition: def,
            name: r?.name ?? ((def.folders.first ?? "") as NSString).lastPathComponent,
            folders: def.folders,
            startDate: start, endDate: end,
            datesOverridden: r?.startDate != nil || r?.endDate != nil,
            places: r?.places ?? detectedPlaces, placesOverridden: r?.places != nil,
            count: stats.count,
            coverHash: r?.coverHash ?? stats.earliestHash, coverOverridden: r?.coverHash != nil)
    }

    /// "YYYY-MM-DD" in the local calendar (the same calendar the timeline groups by).
    public static func dateString(fromMs ms: Int64) -> String {
        let d = Date(timeIntervalSince1970: Double(ms) / 1000)
        let c = Calendar.current.dateComponents([.year, .month, .day], from: d)
        return String(format: "%04d-%02d-%02d", c.year ?? 0, c.month ?? 0, c.day ?? 0)
    }

    /// Newest start first; undated trips last; ties by name (case-insensitive).
    public static func sorted(_ s: [TravelSummary]) -> [TravelSummary] {
        s.sorted { a, b in
            switch (a.startDate, b.startDate) {
            case let (x?, y?) where x != y: return x > y
            case (nil, .some): return false
            case (.some, nil): return true
            default: return a.name.localizedCaseInsensitiveCompare(b.name) == .orderedAscending
            }
        }
    }
}

// Sources/OpenPhotoCore/Catalog/Catalog+Travels.swift
import Foundation
import GRDB

/// A city detected among a trip's photos (from the rebuildable `geocode` table), with its photo count.
public struct DetectedPlace: Sendable, Equatable {
    public let city: String
    public let region: String
    public let country: String
    public let countryCode: String
    public let count: Int
    public init(city: String, region: String, country: String, countryCode: String, count: Int) {
        self.city = city; self.region = region; self.country = country; self.countryCode = countryCode; self.count = count
    }
}

/// Derived facts for a trip = a set of folders, each with its whole subtree. Read-only aggregates
/// over the per-instance browse rows, honouring the locked-folder gate exactly like
/// `items(inDir:recursive:)`. Nothing here is stored — trips derive these at display time.
extension Catalog {
    /// "(dirPath = ? OR dirPath GLOB ?) OR (…)" for every path, with its arguments. GLOB "<p>/*"
    /// matches only paths under "<p>/" so a sibling like "Japan 2026x" never matches "Japan 2026".
    private static func dirClause(_ dirPaths: [String]) -> (sql: String, args: [DatabaseValueConvertible]) {
        var parts: [String] = []; var args: [DatabaseValueConvertible] = []
        for p in dirPaths { parts.append("(dirPath = ? OR dirPath GLOB ?)"); args.append(p); args.append(p + "/*") }
        return (parts.isEmpty ? "0" : "(" + parts.joined(separator: " OR ") + ")", args)
    }

    public func travelStats(dirPaths: [String]) throws -> TravelStats {
        guard !dirPaths.isEmpty else { return TravelStats(count: 0, minTakenAtMs: nil, maxTakenAtMs: nil, earliestHash: nil) }
        let (clause, args) = Self.dirClause(dirPaths)
        return try dbQueue.read { db in
            let base = "SELECT * FROM (\(Self.instanceSQL)) WHERE \(clause) \(lockedFilter)"
            let row = try Row.fetchOne(db, sql: """
                SELECT COUNT(*) AS n, MIN(takenAtMs) AS mn, MAX(takenAtMs) AS mx FROM (\(base))
                """, arguments: StatementArguments(args))
            let n: Int = row?["n"] ?? 0
            guard n > 0 else { return TravelStats(count: 0, minTakenAtMs: nil, maxTakenAtMs: nil, earliestHash: nil) }
            let earliest = try String.fetchOne(db, sql: "SELECT hash FROM (\(base)) ORDER BY takenAtMs ASC, hash ASC LIMIT 1",
                                               arguments: StatementArguments(args))
            return TravelStats(count: n, minTakenAtMs: row?["mn"], maxTakenAtMs: row?["mx"], earliestHash: earliest)
        }
    }

    /// Distinct cities among the trip's photos, most photos first (ties by city name).
    public func travelPlaces(dirPaths: [String]) throws -> [DetectedPlace] {
        guard !dirPaths.isEmpty else { return [] }
        let (clause, args) = Self.dirClause(dirPaths)
        return try dbQueue.read { db in
            try Row.fetchAll(db, sql: """
                SELECT g.city, g.region, g.country, g.countryCode, COUNT(*) AS cnt
                FROM (SELECT hash FROM (\(Self.instanceSQL)) WHERE \(clause) \(lockedFilter)) b
                JOIN geocode g ON g.hash = b.hash
                WHERE g.city <> ''
                GROUP BY g.countryCode, g.city, g.region
                ORDER BY cnt DESC, g.city ASC
                """, arguments: StatementArguments(args)).map {
                DetectedPlace(city: $0["city"], region: $0["region"], country: $0["country"],
                              countryCode: $0["countryCode"], count: $0["cnt"])
            }
        }
    }

    /// The trip's photos, oldest first, one row per physical instance, de-duplicated by instanceID
    /// (a folder listed twice, or nested twice, contributes each file once).
    public func travelItems(dirPaths: [String]) throws -> [TimelineItem] {
        guard !dirPaths.isEmpty else { return [] }
        let (clause, args) = Self.dirClause(dirPaths)
        let rows = try dbQueue.read { db in
            try TimelineItem.fetchAll(db, sql: """
                SELECT * FROM (\(Self.instanceSQL)) WHERE \(clause) \(lockedFilter) ORDER BY takenAtMs ASC, hash ASC
                """, arguments: StatementArguments(args))
        }
        var seen = Set<String>(); return rows.filter { seen.insert($0.instanceID).inserted }
    }
}
```

`lockedFilter` is `"AND locked = 0"` or `""`, which is why it follows the clause with a space. `Row` subscripting to `Int64?` for `mn`/`mx` needs `row?["mn"] as Int64?` if inference complains.

// Sources/OpenPhotoCore/Geocode/PlaceSearch.swift
import Foundation

/// Forward lookup over the bundled GeoNames city list: type a place, get map-recognised
/// `TravelPlace`s (coordinates + GeoNames id). Pure and in-memory, no network. Matching is a
/// case- and diacritic-insensitive prefix match on the name, ASCII name, or any alternate name;
/// exact-name matches rank first, then by population. Absent dataset ⇒ `.empty` (no suggestions).
public struct PlaceSearch: Sendable {
    public struct Entry: Sendable {
        public let geonameID: Int
        public let name: String
        public let asciiName: String
        public let alternateNames: [String]
        public let lat: Double
        public let lon: Double
        public let countryCode: String
        public let region: String
        public let country: String
        public let population: Int
        public init(geonameID: Int, name: String, asciiName: String, alternateNames: [String], lat: Double, lon: Double,
                    countryCode: String, region: String, country: String, population: Int) {
            self.geonameID = geonameID; self.name = name; self.asciiName = asciiName; self.alternateNames = alternateNames
            self.lat = lat; self.lon = lon; self.countryCode = countryCode; self.region = region; self.country = country
            self.population = population
        }
    }

    private let entries: [Entry]
    private let keys: [[String]]          // folded name + ascii + alternates, per entry
    public static let empty = PlaceSearch(entries: [])
    public var isLoaded: Bool { !entries.isEmpty }

    public init(entries: [Entry]) {
        self.entries = entries
        self.keys = entries.map { e in
            var k = [Self.fold(e.name), Self.fold(e.asciiName)]
            k.append(contentsOf: e.alternateNames.map(Self.fold))
            return k
        }
    }

    static func fold(_ s: String) -> String {
        s.folding(options: [.diacriticInsensitive, .caseInsensitive, .widthInsensitive], locale: nil)
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }

    public func suggest(_ query: String, limit: Int = 8) -> [TravelPlace] {
        let q = Self.fold(query)
        guard !q.isEmpty, limit > 0 else { return [] }
        var hits: [(exact: Bool, pop: Int, idx: Int)] = []
        for (i, ks) in keys.enumerated() {
            var exact = false, prefix = false
            for k in ks where k.hasPrefix(q) { prefix = true; if k == q { exact = true; break } }
            if prefix { hits.append((exact, entries[i].population, i)) }
        }
        hits.sort { a, b in
            if a.exact != b.exact { return a.exact }
            if a.pop != b.pop { return a.pop > b.pop }
            return entries[a.idx].name < entries[b.idx].name
        }
        return hits.prefix(limit).map { Self.place(entries[$0.idx]) }
    }

    /// Resolve a detected city (a `geocode` row) to its entry: same country + name; prefer the same
    /// region, else the most populous. Nil if the dataset has no such city.
    public func place(matching d: DetectedPlace) -> TravelPlace? {
        let name = Self.fold(d.city), region = Self.fold(d.region)
        let candidates = entries.enumerated().filter { $0.element.countryCode == d.countryCode && keys[$0.offset][0] == name }
        guard !candidates.isEmpty else { return nil }
        if let same = candidates.first(where: { Self.fold($0.element.region) == region && !region.isEmpty }) {
            return Self.place(same.element)
        }
        return Self.place(candidates.max { $0.element.population < $1.element.population }!.element)
    }

    static func place(_ e: Entry) -> TravelPlace {
        TravelPlace(name: e.name, region: e.region.isEmpty ? nil : e.region, country: e.country.isEmpty ? nil : e.country,
                    countryCode: e.countryCode, latitude: e.lat, longitude: e.lon, geonameID: e.geonameID)
    }
}

// Sources/OpenPhotoCore/Travels/RingGeometry.swift
import Foundation

/// The ring's maths, ported from the browser mockup (mockups/travel-carousel-DISPOSABLE/ring.html):
/// you stand near the centre of a ring of postcards; card i sits at θ = i·step − ringAngle and is placed
/// by translateZ(−pull) · rotateY(θ) · translateZ(−R) · scale(s). The viewer looks along −z from z = +P.
/// A card is hidden only once it is fully outside the stage or about to pass the camera, so cards
/// slide off the edges instead of fading on screen. Pure; the SwiftUI view applies the numbers.
public struct RingGeometry: Sendable, Equatable {
    public var radius: Double = 400
    public var pull: Double = -60
    public var perspective: Double = 950
    public var cardScale: Double = 0.82
    public var cardWidth: Double = 250
    public var cardHeight: Double = 330
    public var edgeMargin: Double = 36
    public init() {}

    public struct CardPlacement: Equatable, Sendable {
        public var angle: Double     // degrees relative to the front (0 = facing you)
        public var x: Double         // projected horizontal offset from the stage centre, before zoom
        public var scale: Double     // projected scale (perspective × card scale × focus bump)
        public var depth: Double     // 0 at the front … 1 at the back (z-order + dimming)
        public var hidden: Bool
    }

    public static func step(count: Int) -> Double { count > 0 ? 360 / Double(count) : 360 }
    public static func wrapDelta(_ d: Double) -> Double {
        var x = (d + 180).truncatingRemainder(dividingBy: 360); if x < 0 { x += 360 }; return x - 180
    }

    /// Projected |x| of a card at 90° — the edge-on point where a card turns and vanishes.
    public var edgeX: Double { radius * perspective / (perspective + pull) }

    /// Smallest zoom that pushes the edge-on point past the stage edges (1 on a very wide stage).
    public func zoom(forStageWidth w: Double) -> Double {
        max(1, (w / 2 + edgeMargin) / edgeX)
    }

    public func placement(index: Int, count: Int, angle: Double, stageWidth: Double) -> CardPlacement {
        let step = Self.step(count: count)
        let th = Self.wrapDelta(Double(index) * step - angle)
        let d = abs(th), rad = th * .pi / 180
        let focus = 1 + max(0, 1 - d / step) * 0.08
        let s = focus * cardScale
        let x = -sin(rad) * radius
        let z = -cos(rad) * radius - pull
        let dist = perspective - z
        let proj = perspective / max(dist, 1)
        let zoom = zoom(forStageWidth: stageWidth)
        let halfW = cardWidth / 2 * s * proj
        let off = (abs(x * proj) - halfW) * zoom > stageWidth / 2 + 8 || dist < 220
        return CardPlacement(angle: th, x: x * proj, scale: s * proj, depth: d / 180, hidden: off)
    }
}

    /// After a folder move/rename `from` → `to`, rewrite every record folder and settings path that is
    /// `from` or lies under it (prefix + "/"), so trips follow their folders. Pure.
    public static func rewritePaths(records: [TravelRecord], settings: TravelSettings, from src: String, to dst: String)
        -> (records: [TravelRecord], settings: TravelSettings, changedRecordIDs: Set<String>) {
        func remap(_ p: String) -> String {
            if p == src { return dst }
            if p.hasPrefix(src + "/") { return dst + String(p.dropFirst(src.count)) }
            return p
        }
        var changed = Set<String>()
        let recs = records.map { r -> TravelRecord in
            var r = r; let f = r.folders.map(remap)
            if f != r.folders { r.folders = f; changed.insert(r.id) }
            return r
        }
        var s = settings
        s.travelsFolder = s.travelsFolder.map(remap)
        s.hidden = s.hidden.map(remap)
        return (recs, s, changed)
    }

        let defs = TravelResolver.resolve(settings: travelSettings, records: TravelStore.loadAll(libraryRoot: root),
                                          folderTree: folderTree)

            let stats = (try? cat.travelStats(dirPaths: def.folders)) ?? TravelStats(count: 0, minTakenAtMs: nil, maxTakenAtMs: nil, earliestHash: nil)
            guard stats.count > 0 else { return nil }          // a trip with no visible photos isn't shown

                placeSearch.place(matching: d) ?? TravelPlace(name: d.city, region: d.region.isEmpty ? nil : d.region,
                                                               country: d.country.isEmpty ? nil : d.country, countryCode: d.countryCode)
            }
            return TravelResolver.summarise(def, stats: stats, detectedPlaces: detected)
        }
        travels = TravelResolver.sorted(sums)

        if selectedTravelID == id { selectedTravelID = rec.id }
        if openedTravelID == id { openedTravelID = rec.id }
        refreshTravels(); refreshToken += 1
    }

import SwiftUI
import AppKit

/// Transparent overlay that forwards trackpad/mouse scroll deltas (SwiftUI has no scroll-wheel
/// gesture on macOS). Sits over the ring stage; clicks pass through (`hitTest` returns nil).
struct ScrollWheelCatcher: NSViewRepresentable {
    var onScroll: (_ dx: CGFloat, _ dy: CGFloat) -> Void

    func makeNSView(context: Context) -> CatcherView { let v = CatcherView(); v.onScroll = onScroll; return v }
    func updateNSView(_ v: CatcherView, context: Context) { v.onScroll = onScroll }

    final class CatcherView: NSView {
        var onScroll: ((CGFloat, CGFloat) -> Void)?
        override func hitTest(_ point: NSPoint) -> NSView? { nil }      // never swallow clicks/drags
        override func viewDidMoveToWindow() {
            super.viewDidMoveToWindow()
            monitor.map(NSEvent.removeMonitor)
            monitor = NSEvent.addLocalMonitorForEvents(matching: .scrollWheel) { [weak self] e in
                guard let self, let win = self.window, e.window === win else { return e }
                let p = self.convert(e.locationInWindow, from: nil)
                guard self.bounds.contains(p) else { return e }
                // Precise (trackpad) deltas are in points; line-based (mouse wheel) are scaled up.
                let k: CGFloat = e.hasPreciseScrollingDeltas ? 1 : 8
                self.onScroll?(e.scrollingDeltaX * k, e.scrollingDeltaY * k)
                return nil                                               // consumed
            }
        }
        private var monitor: Any?
        deinit { monitor.map(NSEvent.removeMonitor) }
    }
}

import AVFoundation
import AppKit

/// The revolver-cylinder click from the mockup: a 35 ms square-wave chirp 1.8 kHz → 0.6 kHz with a
/// fast gain decay, rendered once into a PCM buffer and played through AVAudioEngine. Silent when the
/// system's "Play user interface sound effects" preference is off.
@MainActor final class RatchetSound {
    static let shared = RatchetSound()
    private let engine = AVAudioEngine()
    private let player = AVAudioPlayerNode()
    private var buffer: AVAudioPCMBuffer?

    private init() {
        let sr = 44_100.0
        let format = AVAudioFormat(standardFormatWithSampleRate: sr, channels: 1)!
        let n = Int(sr * 0.035)
        let buf = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: AVAudioFrameCount(n))!
        buf.frameLength = AVAudioFrameCount(n)
        var phase = 0.0
        for i in 0..<n {
            let t = Double(i) / sr
            let f = t < 0.018 ? 1800 * pow(600 / 1800, t / 0.018) : 600           // exponential 1.8k → 0.6k
            phase += 2 * .pi * f / sr
            let gain = 0.05 * pow(0.0005 / 0.05, min(1, t / 0.03))                // 0.05 → 0.0005 over 30 ms
            buf.floatChannelData![0][i] = Float((sin(phase) >= 0 ? 1 : -1) * gain)
        }
        buffer = buf
        engine.attach(player)
        engine.connect(player, to: engine.mainMixerNode, format: format)
        engine.prepare()
    }

    private var uiSoundsEnabled: Bool {
        (UserDefaults.standard.persistentDomain(forName: UserDefaults.globalDomain)?["com.apple.sound.uiaudio.enabled"] as? Int) != 0
    }

    func click() {
        guard uiSoundsEnabled, let buffer else { return }
        if !engine.isRunning { try? engine.start() }
        if !player.isPlaying { player.play() }
        player.scheduleBuffer(buffer, at: nil, options: .interrupts)
    }
}

/// The ring: trips as postcards standing in a circle around you, newest at the front, spun by
/// trackpad scroll, drag, or arrow keys with inertia + snap, and a ratchet click as each card crosses
/// the front. Geometry lives in `RingGeometry` (Core, unit-tested); this view only applies it.
struct TravelRingView: View {
    @Bindable var state: AppState

    @State private var angle = 0.0            // ring rotation, degrees; card i is at i·step − angle
    @State private var velocity = 0.0
    @State private var target: Double?        // snap/step target, eased
    @State private var lastTick = Date()
    @State private var lastClickIndex = -1
    @State private var dragStartX: CGFloat?
    private let geo = RingGeometry()
    private let clock = Timer.publish(every: 1 / 60, on: .main, in: .common).autoconnect()

    private var trips: [TravelSummary] { state.travels }
    private var count: Int { trips.count }
    private var step: Double { RingGeometry.step(count: count) }
    private var focusedIndex: Int { count == 0 ? 0 : ((Int((angle / step).rounded()) % count) + count) % count }
    private var focused: TravelSummary? { trips.isEmpty ? nil : trips[focusedIndex] }

    var body: some View {
        GeometryReader { g in
            let w = g.size.width
            ZStack {
                Theme.windowBG
                if trips.isEmpty { emptyState } else {
                    ring(width: w, height: g.size.height)
                    yearNumeral
                    meta
                    yearStrip
                }
                ScrollWheelCatcher { dx, dy in
                    target = nil
                    let d = abs(dx) > abs(dy) ? dx : dy
                    velocity += Double(d) * 0.045
                }
            }
            .contentShape(Rectangle())
            .gesture(DragGesture(minimumDistance: 3).onChanged { v in
                if dragStartX == nil { dragStartX = v.location.x; target = nil; velocity = 0 }
                let dx = v.location.x - (dragStartX ?? v.location.x); dragStartX = v.location.x
                angle += Double(dx) * -0.16; velocity = Double(dx) * -0.16 * 0.9
            }.onEnded { _ in dragStartX = nil })
            .onReceive(clock) { now in tick(now) }
            .onChange(of: state.selectedTravelID) { _, id in
                if let id, let i = trips.firstIndex(where: { $0.id == id }), i != focusedIndex { stepTo(i) }
            }
            .onChange(of: trips.map(\.id)) { _, _ in
                // The set changed (edit re-sorted it, a trip was added/removed): keep the selected trip in front.
                if let id = state.selectedTravelID, let i = trips.firstIndex(where: { $0.id == id }) {
                    angle = Double(i) * step; target = nil; velocity = 0
                }
            }
        }
        .focusable()
        .focusEffectDisabled()
        .onKeyPress(.rightArrow) { guard !state.isEditingText else { return .ignored }; stepTo((focusedIndex + 1) % max(count, 1)); return .handled }
        .onKeyPress(.leftArrow) { guard !state.isEditingText else { return .ignored }; stepTo((focusedIndex - 1 + count) % max(count, 1)); return .handled }
        .onKeyPress(.return) { guard !state.isEditingText, let f = focused else { return .ignored }; state.openedTravelID = f.id; return .handled }
        .onKeyPress(.init("e")) { guard !state.isEditingText, focused != nil else { return .ignored }; state.travelEditing = true; return .handled }
        .onKeyPress(.escape) { guard !state.isEditingText, state.travelEditing else { return .ignored }; state.travelEditing = false; return .handled }
        .onKeyPress(.home) { stepTo(0); return .handled }
        .onKeyPress(.end) { stepTo(max(count - 1, 0)); return .handled }
    }

    // MARK: physics

    private func tick(_ now: Date) {
        let dt = min(40, now.timeIntervalSince(lastTick) * 1000) / 16.7; lastTick = now
        guard count > 0 else { return }
        if let t = target {
            let d = RingGeometry.wrapDelta(t - angle); angle += d * 0.12 * dt
            if abs(d) < 0.05 { angle = t; target = nil }
        } else {
            angle += velocity * dt; velocity *= pow(0.92, dt)
            if abs(velocity) < 0.08 {
                velocity = 0
                let snap = (angle / step).rounded() * step
                if abs(snap - angle) > 0.01 { target = snap }
            }
        }
        // ratchet: one click per card crossing the front while spinning
        let idx = Int((angle / step).rounded())
        if abs(velocity) > 0.3 && idx != lastClickIndex { lastClickIndex = idx; RatchetSound.shared.click() }
        // publish the focused trip
        if let f = focused, state.selectedTravelID != f.id, target == nil || abs(RingGeometry.wrapDelta((target ?? angle) - angle)) < step / 2 {
            state.selectedTravelID = f.id
        }
    }

    private func stepTo(_ i: Int) { guard count > 0 else { return }; target = angle + RingGeometry.wrapDelta(Double(i) * step - angle) }

    // MARK: ring

    private func ring(width: CGFloat, height: CGFloat) -> some View {
        let zoom = geo.zoom(forStageWidth: Double(width))
        return ZStack {
            ForEach(Array(trips.enumerated()), id: \.element.id) { i, trip in
                let p = geo.placement(index: i, count: count, angle: angle, stageWidth: Double(width))
                if !p.hidden {
                    card(trip, focused: i == focusedIndex)
                        .frame(width: geo.cardWidth, height: geo.cardHeight)
                        .rotation3DEffect(.degrees(p.angle), axis: (x: 0, y: 1, z: 0), perspective: 0.6)
                        .scaleEffect(p.scale)
                        .offset(x: p.x)
                        .zIndex(1 - p.depth)
                        .onTapGesture { if i == focusedIndex { state.openedTravelID = trip.id } else { stepTo(i) } }
                }
            }
        }

    }

            LinearGradient(colors: [.black.opacity(0.72), .clear], startPoint: .bottom, endPoint: .center)
            VStack(alignment: .leading, spacing: 2) {

        .overlay(alignment: .topTrailing) {
            Text(trip.count.formatted()).font(.system(size: 10.5).monospacedDigit()).foregroundStyle(.white)
                .padding(.horizontal, 7).padding(.vertical, 2).background(.black.opacity(0.5), in: Capsule()).padding(10)
        }
        .clipShape(RoundedRectangle(cornerRadius: 8))
        .overlay(RoundedRectangle(cornerRadius: 8).strokeBorder(focused ? Theme.accent : .white.opacity(0.08), lineWidth: focused ? 2 : 1))
        .shadow(color: .black.opacity(0.45), radius: 30, y: 30)
    }

    /// Cover image for a trip (its `coverHash`), by content hash from the thumbnail store.

/// Human formatting shared by the ring, inspector and trip page.
enum TravelFormat {
    static func date(_ s: String) -> Date? {
        let p = s.split(separator: "-").compactMap { Int($0) }; guard p.count == 3 else { return nil }
        return Calendar.current.date(from: DateComponents(year: p[0], month: p[1], day: p[2]))
    }
    static func days(_ t: TravelSummary) -> Int {
        guard let a = t.startDate.flatMap(date), let b = t.endDate.flatMap(date) else { return 0 }
        return (Calendar.current.dateComponents([.day], from: a, to: b).day ?? 0) + 1
    }
    /// "3 Apr – 15 Apr 2026 · 13 days"
    static func span(_ t: TravelSummary) -> String {
        guard let a = t.startDate.flatMap(date), let b = t.endDate.flatMap(date) else { return "no dates" }
        let f = Date.FormatStyle().month(.abbreviated).day()
        let ya = Calendar.current.component(.year, from: a), yb = Calendar.current.component(.year, from: b)
        let y = ya == yb ? String(yb) : "\(ya)–\(yb)"
        let n = days(t)
        return "\(a.formatted(f)) – \(b.formatted(f)) \(y) · \(n) day\(n == 1 ? "" : "s")"
    }
}

/// Place chips + a type-to-add field with GeoNames suggestions. Return picks the highlighted

                    }
                }
                .padding(4).background(Theme.elevated, in: RoundedRectangle(cornerRadius: 6))
                .overlay(RoundedRectangle(cornerRadius: 6).strokeBorder(Theme.hairline))
            }
        }
    }

    private func chip(_ p: TravelPlace, index: Int) -> some View {
        HStack(spacing: 4) {
            Text(p.name).font(.system(size: 12, weight: .medium))
            Button { var n = places; n.remove(at: index); onChange(n) } label: { Image(systemName: "xmark").font(.system(size: 8, weight: .bold)) }
                .buttonStyle(.plain).opacity(0.7)
        }
        .padding(.leading, 9).padding(.trailing, 6).padding(.vertical, 3)
        .background(overridden && p.isOnMap ? Theme.accentDim : Theme.elevated, in: Capsule())
        .overlay(Capsule().strokeBorder(p.isOnMap ? Theme.hairline : Theme.amber.opacity(0.7), style: StrokeStyle(lineWidth: 1, dash: p.isOnMap ? [] : [3, 2])))
        .foregroundStyle(overridden && p.isOnMap ? Theme.accent : Theme.text)
        .help(p.isOnMap ? [p.region, p.country].compactMap { $0 }.joined(separator: ", ") : "not on the map")
    }

    private func commit() {
        let t = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !t.isEmpty else { return }
        let pick = suggestions.indices.contains(highlighted) ? suggestions[highlighted] : TravelPlace(name: t)
        if !places.contains(where: { $0.geonameID != nil && $0.geonameID == pick.geonameID || $0.name.caseInsensitiveCompare(pick.name) == .orderedSame }) {
            onChange(places + [pick])
        }
        text = ""; suggestions = []
    }
}

/// Minimal wrapping HStack for chips.
struct FlowLayout: Layout {
    var spacing: CGFloat = 6
    func sizeThatFits(proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) -> CGSize {
        let w = proposal.width ?? 300; var x: CGFloat = 0, y: CGFloat = 0, rowH: CGFloat = 0
        for s in subviews {
            let sz = s.sizeThatFits(.unspecified)
            if x + sz.width > w && x > 0 { x = 0; y += rowH + spacing; rowH = 0 }
            x += sz.width + spacing; rowH = max(rowH, sz.height)
        }
        return CGSize(width: w, height: y + rowH)
    }
    func placeSubviews(in b: CGRect, proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) {
        var x = b.minX, y = b.minY, rowH: CGFloat = 0
        for s in subviews {
            let sz = s.sizeThatFits(.unspecified)
            if x + sz.width > b.maxX && x > b.minX { x = b.minX; y += rowH + spacing; rowH = 0 }
            s.place(at: CGPoint(x: x, y: y), proposal: ProposedViewSize(sz))
            x += sz.width + spacing; rowH = max(rowH, sz.height)
        }
    }
}

        ZStack(alignment: .bottomLeading) {
            if let lib = state.library { TravelRingView.coverThumbnail(for: trip, library: lib, targetPixel: 1600).frame(height: 300).clipped() }
            LinearGradient(colors: [Theme.windowBG, .clear], startPoint: .bottom, endPoint: .init(x: 0.5, y: 0.4))
            VStack(alignment: .leading, spacing: 6) {
                Text(trip.name).font(.system(size: 34, weight: .semibold)).kerning(-0.8)
                Text("\(TravelFormat.span(trip)) · \(trip.places.map(\.name).joined(separator: " · "))").font(.system(size: 14)).foregroundStyle(Theme.textDim)
            }.padding(.horizontal, 28).padding(.bottom, 18)
        }
        .frame(height: 300)
        .overlay(alignment: .top) {
            HStack {
                Button { state.openedTravelID = nil } label: { Label("Travels", systemImage: "chevron.left") }
                Spacer()
                Button("Edit trip") { state.openedTravelID = nil; state.travelEditing = true }
            }.controlSize(.small).padding(14)
        }
    }

    private func fact(_ label: String, _ value: String) -> some View {
        VStack(alignment: .leading, spacing: 3) {
            Text(label.uppercased()).font(.system(size: 11, weight: .semibold)).kerning(0.66).foregroundStyle(Theme.textFaint)
            Text(value).font(.system(size: 13.5).monospacedDigit())
        }
    }

    @ViewBuilder private func tile(_ item: TimelineItem) -> some View {
        if let lib = state.library {
            MediaTile(id: item.instanceID,
                      thumbnail: ThumbnailImage(timelineItem: item, library: lib, targetPixel: thumbPixels),
                      badges: { TimelineTileBadges(item: item, backedUp: state.isBackedUpOnCanonical(item)) },
                      onTap: { state.openViewer(item, within: all) })
                .contextMenu { Button("Set as Trip Cover", systemImage: "photo") { state.setTravelCover(id: trip.id, item.hash) } }
        }
    }

    /// Group by local calendar day (oldest first) and pick the day's most common geocoded city.
    private func load() {
        all = state.travelItems(trip)
        let cal = Calendar.current
        var groups: [(Date, [TimelineItem])] = []
        for it in all {
            let d = cal.startOfDay(for: Date(timeIntervalSince1970: Double(it.takenAtMs) / 1000))
            if let last = groups.last, last.0 == d { groups[groups.count - 1].1.append(it) } else { groups.append((d, [it])) }
        }
        let f = Date.FormatStyle().weekday(.wide).month(.wide).day()
        days = groups.map { day, items in
            var counts: [String: Int] = [:]
            if let cat = state.library?.catalog {
                for it in items.prefix(200) { if let g = try? cat.geocode(forHash: it.hash), !g.city.isEmpty { counts[g.city, default: 0] += 1 } }
            }
            let place = counts.max { $0.value < $1.value }?.key
            return (day.formatted(f), place, items)
        }
    }
}
